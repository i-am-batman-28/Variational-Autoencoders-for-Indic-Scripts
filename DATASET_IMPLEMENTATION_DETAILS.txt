STEP 3: Base Image Rendering
-----------------------------
We rendered each character at specified point size with 300 DPI:

```python
def render_character(char, font_path, size_pt, dpi=300):
    """
    Render a single character to image.
    
    Parameters:
    - char: Unicode character string
    - font_path: Path to .ttf font file
    - size_pt: Font size in points (10, 14, 18, or 22)
    - dpi: Dots per inch (resolution)
    
    Returns:
    - PIL Image object (RGB, white background, black text)
    """
    
    # Calculate font size in pixels
    # Formula: pixels = points * dpi / 72
    font_size_px = int(size_pt * dpi / 72)
    
    # Load font at calculated size
    font = ImageFont.truetype(font_path, size=font_size_px)
    
    # Create temporary large image to get bounding box
    temp_img = Image.new('RGB', (500, 500), color='white')
    temp_draw = ImageDraw.Draw(temp_img)
    
    # Get text bounding box
    bbox = temp_draw.textbbox((0, 0), char, font=font)
    text_width = bbox[2] - bbox[0]
    text_height = bbox[3] - bbox[1]
    
    # Add padding (20% on each side)
    padding = int(max(text_width, text_height) * 0.2)
    canvas_size = max(text_width, text_height) + 2 * padding
    
    # Create final image with appropriate size
    image = Image.new('RGB', (canvas_size, canvas_size), color='white')
    draw = ImageDraw.Draw(image)
    
    # Calculate centered position
    x = (canvas_size - text_width) // 2 - bbox[0]
    y = (canvas_size - text_height) // 2 - bbox[1]
    
    # Draw character
    draw.text((x, y), char, font=font, fill='black')
    
    # Resize to standard 128x128
    image = image.resize((128, 128), Image.LANCZOS)
    
    return image
```

TECHNICAL DETAILS:
- DPI = 300 (print quality, ensures crisp edges)
- Resampling: LANCZOS (highest quality downsampling filter)
- Background: RGB(255, 255, 255) - pure white
- Foreground: RGB(0, 0, 0) - pure black
- Auto-centering: Bounding box calculation ensures character is centered
- Padding: 20% around character to avoid edge clipping

TRANSFORMATION 1: ROTATION

```python
import random
from PIL import Image

def apply_rotation(image, variation_seed):
    """
    Apply random rotation between -3° to +3°.
    
    Parameters:
    - image: PIL Image
    - variation_seed: Integer seed for reproducibility
    
    Returns:
    - Rotated PIL Image
    """
    random.seed(variation_seed)
    
    # Random angle in range [-3, +3] degrees
    angle = random.uniform(-3.0, 3.0)
    
    # Rotate with bicubic interpolation, expand canvas to avoid cropping
    rotated = image.rotate(
        angle, 
        resample=Image.BICUBIC,
        expand=False,  # Keep 128x128 size
        fillcolor='white'  # Fill exposed areas with white
    )
    
    return rotated, {'rotation_deg': round(angle, 2)}
```

PARAMETERS:
- Range: [-3°, +3°]
- Distribution: Uniform random
- Interpolation: Bicubic (smooth, high-quality)
- Fill color: White (maintains background)

TRANSFORMATION 2: TRANSLATION
-----------------------------
```python
def apply_translation(image, variation_seed):
    """
    Apply random translation ±3 pixels in X and Y.
    
    Parameters:
    - image: PIL Image
    - variation_seed: Integer seed
    
    Returns:
    - Translated PIL Image
    """
    random.seed(variation_seed + 1000)  # Different seed than rotation
    
    # Random shifts
    shift_x = random.randint(-3, 3)  # pixels
    shift_y = random.randint(-3, 3)  # pixels
    
    # Apply translation using affine transform
    # Affine matrix: [1, 0, shift_x, 0, 1, shift_y]
    translated = image.transform(
        image.size,
        Image.AFFINE,
        (1, 0, shift_x, 0, 1, shift_y),
        resample=Image.BICUBIC,
        fillcolor='white'
    )
    
    return translated, {
        'translation_x_px': shift_x,
        'translation_y_px': shift_y
    }
```

PARAMETERS:
- Range X: [-3, +3] pixels
- Range Y: [-3, +3] pixels
- Distribution: Uniform random (discrete integers)
- At 128×128: 3px = 2.34% of image width

TRANSFORMATION 3: SCALING
-------------------------
```python
def apply_scaling(image, variation_seed):
    """
    Apply random scaling between 94% to 106%.
    
    Parameters:
    - image: PIL Image
    - variation_seed: Integer seed
    
    Returns:
    - Scaled PIL Image
    """
    random.seed(variation_seed + 2000)
    
    # Random scale factor
    scale_factor = random.uniform(0.94, 1.06)
    
    # Calculate new size
    new_size = int(128 * scale_factor)
    
    # Resize
    scaled = image.resize((new_size, new_size), Image.LANCZOS)
    
    # Paste back into 128x128 canvas (centered)
    canvas = Image.new('RGB', (128, 128), color='white')
    offset = (128 - new_size) // 2
    canvas.paste(scaled, (offset, offset))
    
    return canvas, {'scale_factor': round(scale_factor, 3)}
```

PARAMETERS:
- Range: [0.94, 1.06] (94% to 106%)
- Distribution: Uniform random (continuous)
- Resampling: LANCZOS
- Effect: ±6% size variation

TRANSFORMATION 4: STROKE MODIFICATION (Morphological Operations)
-----------------------------------------------------------------
```python
import cv2
import numpy as np

def apply_stroke_modification(image, variation_seed):
    """
    Apply erosion (thinning) or dilation (thickening) to strokes.
    
    Parameters:
    - image: PIL Image
    - variation_seed: Integer seed
    
    Returns:
    - Modified PIL Image
    """
    random.seed(variation_seed + 3000)
    
    # Convert PIL to numpy array
    img_array = np.array(image)
    
    # Define morphological kernel (2x2 square)
    kernel = np.ones((2, 2), np.uint8)
    
    # Random choice: erosion, dilation, or none
    operation = random.choice(['erode', 'dilate', 'none'])
    
    if operation == 'erode':
        # Thinning strokes
        modified = cv2.erode(img_array, kernel, iterations=1)
    elif operation == 'dilate':
        # Thickening strokes
        modified = cv2.dilate(img_array, kernel, iterations=1)
    else:
        # No modification
        modified = img_array
    
    # Convert back to PIL
    result = Image.fromarray(modified)
    
    return result, {'morphology': operation}
```

PARAMETERS:
- Operations: Erosion, Dilation, or None
- Kernel: 2×2 square structuring element
- Iterations: 1 (conservative, preserves character)
- Distribution: Each operation has 33% probability

TRANSFORMATION 5: COMBINED APPLICATION
---------------------------------------
```python
def generate_variation(base_image, variation_number):
    """
    Apply all transformations to create one variation.
    
    Parameters:
    - base_image: Original rendered character (PIL Image)
    - variation_number: 1-20 (used as seed)
    
    Returns:
    - transformed_image: PIL Image
    - metadata: Dictionary of all applied transformations
    """
    
    # Use variation_number as base seed for reproducibility
    seed = variation_number
    
    # Apply transformations in sequence
    img = base_image.copy()
    metadata = {}
    
    # 1. Rotation
    img, rot_meta = apply_rotation(img, seed)
    metadata.update(rot_meta)
    
    # 2. Translation
    img, trans_meta = apply_translation(img, seed)
    metadata.update(trans_meta)
    
    # 3. Scaling
    img, scale_meta = apply_scaling(img, seed)
    metadata.update(scale_meta)
    
    # 4. Stroke modification
    img, stroke_meta = apply_stroke_modification(img, seed)
    metadata.update(stroke_meta)
    
    return img, metadata
```

IMPORTANT: Transformation Order Matters!
- Rotation → Translation → Scaling → Stroke
- This order ensures geometric transforms happen before pixel-level operations

1.3 QUALITY CONTROL - SHA-256 DEDUPLICATION

We implemented cryptographic hashing to detect duplicate images:

```python
import hashlib

def compute_image_hash(image):
    """
    Compute SHA-256 hash of image data.
    
    Parameters:
    - image: PIL Image
    
    Returns:
    - 64-character hexadecimal hash string
    """
    # Convert image to bytes
    img_bytes = image.tobytes()
    
    # Compute SHA-256 hash
    hash_obj = hashlib.sha256(img_bytes)
    hash_hex = hash_obj.hexdigest()
    
    return hash_hex

# Deduplication tracking
hash_set = set()
duplicate_count = 0

for image_file, image in generated_images.items():
    hash_val = compute_image_hash(image)
    
    if hash_val in hash_set:
        print(f"DUPLICATE FOUND: {image_file}")
        duplicate_count += 1
    else:
        hash_set.add(hash_val)

print(f"Total duplicates: {duplicate_count}")  # Result: 0
```

WHY SHA-256?
- Extremely low collision probability (< 10^-70)
- Industry standard for data integrity
- Fast computation (~0.001 seconds per image)
- Produces unique 256-bit fingerprint

```python
def normalize_images(images):
    """
    Normalize pixel values to [0, 1] range.
    
    Parameters:
    - images: NumPy array (N, 128, 128, 3) with dtype uint8
    
    Returns:
    - normalized_images: NumPy array (N, 128, 128, 3) with dtype float32
    """
    
    # Convert to float32 and normalize
    normalized = images.astype('float32') / 255.0
    
    # Verify range
    assert normalized.min() >= 0.0, "Minimum should be >= 0"
    assert normalized.max() <= 1.0, "Maximum should be <= 1"
    
    print(f"Normalization complete")
    print(f"  Min value: {normalized.min():.4f}")
    print(f"  Max value: {normalized.max():.4f}")
    print(f"  Mean value: {normalized.mean():.4f}")
    
    return normalized

# Apply normalization
normalized_images = normalize_images(images)

print(f"Normalized shape: {normalized_images.shape}")   # (4160, 128, 128, 3)
print(f"Normalized dtype: {normalized_images.dtype}")    # float32
```

WHY NORMALIZE?
1. Neural networks train better with inputs in [0, 1] or [-1, 1]
2. Prevents gradient explosion (large pixel values → large gradients)
3. Improves optimizer convergence (Adam, SGD work better)
4. Standardizes input scale across different images

ALTERNATIVE: Standardization (zero mean, unit variance)
```python
def standardize_images(images):
    """Alternative: Z-score normalization"""
    mean = images.mean()
    std = images.std()
    standardized = (images - mean) / std
    return standardized
```
We chose [0, 1] normalization as it's standard for image generation tasks.

We check for corrupted images (all-white, all-black, or zero variance):

```python
def sanity_check_images(images, min_variance=0.001):
    """
    Detect corrupted images based on pixel statistics.
    
    Parameters:
    - images: NumPy array (N, 128, 128, 3), normalized to [0, 1]
    - min_variance: Minimum acceptable variance
    
    Returns:
    - valid_indices: Indices of valid images
    """
    
    valid_indices = []
    issues = {
        'all_white': 0,
        'all_black': 0,
        'low_variance': 0
    }
    
    for i, img in enumerate(images):
        # Compute statistics
        mean_val = img.mean()
        variance = img.var()
        
        # Check for all-white (mean ≈ 1.0)
        if mean_val > 0.98:
            issues['all_white'] += 1
            continue
        
        # Check for all-black (mean ≈ 0.0)
        if mean_val < 0.02:
            issues['all_black'] += 1
            continue
        
        # Check for low variance (no structure)
        if variance < min_variance:
            issues['low_variance'] += 1
            continue
        
        # Image is valid
        valid_indices.append(i)
    
    print(f"\nContent sanity check:")
    print(f"  Valid images: {len(valid_indices)}")
    print(f"  All-white images: {issues['all_white']}")
    print(f"  All-black images: {issues['all_black']}")
    print(f"  Low-variance images: {issues['low_variance']}")
```

We use stratified splitting to ensure balanced class distribution:

```python
from sklearn.model_selection import train_test_split
import numpy as np

def create_train_test_split(images, metadata, test_size=0.2, random_state=42):
    """
    Split dataset into training and testing sets with stratification.
    
    Parameters:
    - images: NumPy array (N, 128, 128, 3)
    - metadata: List of metadata dictionaries
    - test_size: Fraction for test set (0.2 = 20%)
    - random_state: Random seed for reproducibility
    
    Returns:
    - X_train, X_test, meta_train, meta_test
    """
    
    # Extract labels for stratification (character + font)
    labels = []
    for meta in metadata:
        label = f"{meta['script']}_{meta['character_name']}_{meta['font_name']}"
        labels.append(label)
    
    labels = np.array(labels)
    
    print(f"\nClass distribution:")
    unique, counts = np.unique(labels, return_counts=True)
    for label, count in zip(unique[:5], counts[:5]):  # Show first 5
        print(f"  {label}: {count} samples")
    print(f"  ... ({len(unique)} unique classes total)")
    
    # Stratified split
    X_train, X_test, y_train, y_test, meta_train, meta_test = train_test_split(
        images, 
        labels,
        metadata,
        test_size=test_size,
        random_state=random_state,
        stratify=labels  # Ensures balanced split
    )
    
    print(f"\nTrain-Test Split:")
    print(f"  Training samples: {len(X_train)} ({100*(1-test_size):.0f}%)")
    print(f"  Testing samples: {len(X_test)} ({100*test_size:.0f}%)")
    
    # Verify class balance in both sets
    train_unique, train_counts = np.unique(y_train, return_counts=True)
    test_unique, test_counts = np.unique(y_test, return_counts=True)
    
    print(f"  Classes in train set: {len(train_unique)}")
    print(f"  Classes in test set: {len(test_unique)}")
    print(f"  Balance verified: {len(train_unique) == len(test_unique)}")
    
    return X_train, X_test, meta_train, meta_test

# Create split

    normalized_images,
```

```python
import tensorflow as tf

def create_tf_dataset(X_train, X_test, batch_size=32, shuffle=True):
    """
    Create TensorFlow datasets for efficient training.
    
    Parameters:
    - X_train: NumPy array (N_train, 128, 128, 3)
    - X_test: NumPy array (N_test, 128, 128, 3)
    - batch_size: Number of samples per batch
    - shuffle: Whether to shuffle training data
    
    Returns:
    - train_dataset: tf.data.Dataset
    - test_dataset: tf.data.Dataset
    """
    
    # Create training dataset
    train_dataset = tf.data.Dataset.from_tensor_slices(X_train)
    
    if shuffle:
        # Shuffle buffer size = size of training set
        train_dataset = train_dataset.shuffle(
            buffer_size=len(X_train),
            seed=42,
            reshuffle_each_iteration=True  # Re-shuffle every epoch
        )
    
    train_dataset = train_dataset.batch(batch_size)
    train_dataset = train_dataset.prefetch(tf.data.AUTOTUNE)  # Optimize loading
    
    # Create test dataset (no shuffling)
    test_dataset = tf.data.Dataset.from_tensor_slices(X_test)
    test_dataset = test_dataset.batch(batch_size)
    test_dataset = test_dataset.prefetch(tf.data.AUTOTUNE)
    
    print(f"\nDataset batching:")
    print(f"  Batch size: {batch_size}")
    print(f"  Training batches: {len(X_train) // batch_size}")
    print(f"  Testing batches: {len(X_test) // batch_size}")
    
    return train_dataset, test_dataset

# Create datasets
train_dataset, test_dataset = create_tf_dataset(
    X_train, 
    X_test, 
    batch_size=32,
    shuffle=True
)
```

BATCHING DETAILS:
- Batch size: 32 (balances memory and convergence)
- Training batches: 3,328 / 32 = 104 batches per epoch
- Testing batches: 832 / 32 = 26 batches
- Shuffle buffer: 3,328 (entire training set)
- Prefetching: Overlaps data loading with computation (faster training)
